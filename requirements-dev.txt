# Development and test dependencies, on top of requirements.txt.
#     .venv\Scripts\python.exe -m pip install -r requirements-dev.txt
#     .venv\Scripts\python.exe -m playwright install --only-shell
#
# Playwright drives the headless browser checks in
# tests/test_web_gui_browser.py and scripts/capture_screenshots.py. Those tests
# skip themselves when it is missing, so the rest of the suite still runs.
#
# "--only-shell" fetches just chromium-headless-shell (~270 MB) instead of the
# full chromium build (~410 MB on top of it). Both callers launch with
# channel="chromium-headless-shell"; Playwright does not fall back from the
# chromium executable to the shell, so keep the channel and the install flag in
# sync if either changes.
